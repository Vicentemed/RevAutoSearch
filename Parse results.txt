parse_serpapi_results <- function(json_raw) {

  items <- json_raw$organic_results
  if (is.null(items) || length(items) == 0) return(data.frame())

  # Garantiza que cada item sea lista
  items <- lapply(items, function(x) {
    if (!is.list(x)) return(list())  # si viene atómico -> list vacía
    x
  })

  safe_val <- function(x) {
    if (is.null(x) || length(x) == 0) return("")
    if (is.atomic(x)) return(as.character(x))
    return(as.character(x))
  }

  titles <- sapply(items, function(x) safe_val(x[["title"]]))
  snippets <- sapply(items, function(x) safe_val(x[["snippet"]]))
  links <- sapply(items, function(x) safe_val(x[["link"]]))

  years <- sapply(items, function(x) {
    pi <- x[["publication_info"]]
    if (is.null(pi) || !is.list(pi)) return("")
    yr <- pi[["year"]]
    if (is.null(yr)) "" else as.character(yr)
  })

  df <- data.frame(
    title_raw = titles,
    snippet   = snippets,
    link      = links,
    year_raw  = years,
    stringsAsFactors = FALSE
  )

  df$title_clean <- sapply(df$title_raw, normalize_title)

  df
}
